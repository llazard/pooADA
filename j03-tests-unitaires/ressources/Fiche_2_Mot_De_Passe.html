Question 0, avec la formatrice

Copiez les deux fichiers, lancez PasswordTest : deux tests verts. Puis lancez le avec la couverture : clic droit sur la classe, « More Run/Debug », « Run with Coverage ». Dans le chat : combien de lignes de check restent rouges ?

import java.util.ArrayList;
import java.util.List;

public class Password {

    // Renvoie la liste des règles non respectées. Liste vide : le mot de passe est valide.
    public static List<String> check(String password) {
        List<String> errors = new ArrayList<>();

        if (password.length() < 12) {
            errors.add("Au moins 12 caractères");
        }

        boolean hasLetter = false;
        boolean hasDigit = false;
        for (char c : password.toCharArray()) {
            if (Character.isLetter(c)) {
                hasLetter = true;
            }
            if (Character.isDigit(c)) {
                hasDigit = true;
            }
        }
        if (!hasLetter) {
            errors.add("Au moins une lettre");
        }

        return errors;
    }
}
import org.junit.jupiter.api.Test;

import java.util.List;

import static org.junit.jupiter.api.Assertions.assertEquals;
import static org.junit.jupiter.api.Assertions.assertTrue;

class PasswordTest {

    @Test
    void check_validPassword_noError() {
        List<String> errors = Password.check("Azerty123!xyz");
        assertTrue(errors.isEmpty());
    }

    @Test
    void check_tooShort_oneError() {
        List<String> errors = Password.check("Azerty1!");
        assertEquals(1, errors.size());
        assertTrue(errors.contains("Au moins 12 caractères"));
    }
}
